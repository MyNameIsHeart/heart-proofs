\part{פונקציות קדומות )אינטגרל לא מסוים(}

הנימוק \L{$\text{linearity}$} מתייחס ללינאריות האינטגרל הלא מסוים.
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{1}{3x+4}dx
\]
}%
 עבור \L{$x>0$}.
\end{problem*}
\begin{sol*}
על מנת לפתור את הבעיה, נשתמש בכתיב לייבניץ לנגזרת ובמשפט אינטגרציה
בהצבה.

נזהה שהאינטגרל הלא מסוים \L{$\int\frac{1}{x}dx$} הוא קל לחישוב ודומה
למבנה האינטגרל הלא מסוים שלנו, על כן נגדיר \L{%
\[
u=3x+4
\]
}%
בשביל להציב את \L{$u$} באינטגרל הלא מסוים שלנו, נרצה לחשב את האינטגרל
הלא מסוים לפי \L{$u$} ולא לפי \L{$x$}.

לשם כך, נצטרך לבטא את \L{$dx$} בעזרת \L{$du$}:\L{%
\[
\frac{du}{dx}=u'=3
\]
}%
ועל כן\L{%
\[
dx=\frac{1}{3}du
\]
}%
לכן נוכל לכתוב\L{%
\[
\int\frac{1}{3x+4}dx=\int\frac{1}{u}\ \frac{1}{3}du\underset{\text{linearity}}{=}\frac{1}{3}\int\frac{1}{u}\ du=\frac{1}{3}\ln u+C=\frac{1}{3}\ln\left(3x+4\right)+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{1}{4x^{2}+12x+10}dx
\]
}%
\end{problem*}
\begin{sol*}
על מנת לפתור את הבעיה, נשתמש בכתיב לייבניץ לנגזרת ובמשפט אינטגרציה
בהצבה.

נזהה שהאינטגרל הלא מסוים \L{$\int\frac{1}{x^{2}+1}dx$} הוא קל לחישוב
ודומה למבנה האינטגרל הלא מסוים שלנו - על כן ננסה \textquotedblleft לייצר\textquotedblright{}
אותו.\\
\noindent\fbox{\begin{minipage}[t]{1\columnwidth - 2\fboxsep - 2\fboxrule}%
\begin{center}
טיוטה:\L{%
\[
mn=4\cdot9=36
\]
}%
\L{%
\[
m+n=12
\]
}%
\L{%
\[
m=n=6
\]
}%
\L{%
\[
4x^{2}+12x+9=\left(2x+3\right)^{2}
\]
}%
\par\end{center}%
\end{minipage}}

\L{%
\[
\frac{1}{4x^{2}+12x+10}=\frac{1}{4x^{2}+12x+9+1}=\frac{1}{\left(2x+3\right)^{2}+1}
\]
}%

ולכן נגדיר \L{%
\[
u=2x+3
\]
}%
\L{%
\[
\frac{du}{dx}=2\Rightarrow dx=\frac{1}{2}du
\]
}%
\L{%
\[
\therefore\int\frac{1}{4x^{2}+12x+10}dx\underset{\text{linearity}}{=}\frac{1}{2}\int\frac{1}{u^{2}+1}\ du=\frac{1}{2}\arctan\left(u\right)+C=\frac{1}{2}\arctan\left(2x+3\right)+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים\L{%
\[
\int\sin x\cos x\ dx
\]
}%
\end{problem*}
\begin{sol*}
נשים לב שמתקיים\L{%
\[
\int\sin x\cos x\ dx=\int\sin x\sin'x\ dx
\]
}%
נציב \L{$u=\sin x$} ונקבל\L{%
\[
\frac{du}{dx}=\sin'x\Rightarrow du=\sin'x\ dx
\]
}%
\L{%
\[
\therefore\int\sin x\sin'x\ dx=\int u\ du=\frac{u^{2}}{2}+C=\frac{\sin^{2}x}{2}+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים\L{%
\[
\int\sin^{6}(x)\cos^{5}(x)\ dx
\]
}%
\end{problem*}
\begin{sol*}
כמו שראינו בשאלה הקודמת, אם נוכל לייצר מבנה שבתוכו יש \L{$\sin(x)$}
והוא מסתיים ב- \L{$\cos(x)\ dx$} נוכל לבחור \L{$\sin(x)=u$}.

לכן, נרצה להשאיר \L{$\cos(x)$} אחד, ולהוריד את הדרגה של הפונקציות
האחרות.\L{%
\[
\sin^{2}(x)+\cos^{2}(x)=1\Rightarrow\cos^{2}(x)=1-\sin^{2}(x)
\]
}%
\L{%
\[
\therefore\int\sin^{6}(x)\cos^{5}(x)\ dx=\int\sin^{6}(x)\cos^{4}(x)\cos(x)\ dx=\int\sin^{6}(x)\left(1-\sin^{2}(x)\right)^{2}\cos(x)\ dx
\]
}%
נבחר \L{$u=\sin(x)$} לכן \L{$du=\cos(x)\ dx$} ונקבל\L{%
\[
=\int u^{6}\left(1-u^{2}\right)^{2}\ du=\int u^{6}\left(1-2u^{2}+u^{4}\right)\ du=\int u^{6}-2u^{8}+u^{10}\ du
\]
}%
\L{%
\[
\underset{\text{linearity}}{=}\int u^{6}\ du-2\int u^{8}\ du+\int u^{10}\ du=\frac{u^{7}}{7}-\frac{2u^{9}}{9}+\frac{u^{11}}{11}+C
\]
}%
\L{%
\[
=\frac{\sin(x)^{7}}{7}-\frac{2\sin(x)^{9}}{9}+\frac{\sin(x)^{11}}{11}+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים\L{%
\[
\int2xe^{\left(x^{2}\right)}\ dx
\]
}%
\end{problem*}
\begin{sol*}
נשים לב שמתקיים\L{%
\[
\left(x^{2}\right)^{'}=2x,\quad\left(e^{x}\right)^{'}=e^{x}
\]
}%

ולכן ישירות לפי אינטגרציה בהצבה\L{%
\[
\int2xe^{\left(x^{2}\right)}\ dx=\int e^{\left(x^{2}\right)}\cdot\left(x^{2}\right)^{'}\ dx=e^{\left(x^{2}\right)}+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים\L{%
\[
\int e^{x}\sin(x)\ dx
\]
}%
\end{problem*}
\begin{sol*}
זה מבנה שמתאים לאינטגרציה בחלקים: פונקציה שקל לטנגרל \L{$\left(e^{x}\right)$}
כפול פונקציה שקל לגזור \L{$\left(\sin x\right)$}. \L{%
\[
\int e^{x}\sin(x)\ dx=\int\sin(x)\cdot\left(e^{x}\right)^{'}\ dx=\sin(x)\cdot e^{x}-\int\cos(x)\cdot e^{x}\ dx
\]
}%

באופן דומה\L{%
\[
\int\cos(x)\cdot e^{x}\ dx=\int\cos(x)\cdot\left(e^{x}\right)'\ dx=\cos(x)\cdot e^{x}-\int-\sin(x)\cdot e^{x}\ dx
\]
}%

ובסך הכל קיבלנו\L{%
\[
\int e^{x}\sin(x)\ dx=\sin(x)\cdot e^{x}-\cos(x)\cdot e^{x}-\int\sin(x)\cdot e^{x}\ dx
\]
}%
\L{%
\[
2\int e^{x}\sin(x)\ dx=\sin(x)\cdot e^{x}-\cos(x)\cdot e^{x}+C
\]
}%
\L{%
\[
\int e^{x}\sin(x)\ dx=\frac{\sin(x)\cdot e^{x}-\cos(x)\cdot e^{x}}{2}+C
\]
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{\sin x}{1+\cos^{2}x}\,dx
\]
}%
\end{problem*}
\begin{sol*}
נציב \L{%
\[
u=-\cos(x),\qquad\frac{du}{dx}=\sin(x)
\]
}%
\L{%
\begin{align*}
\int\frac{\sin x}{1+\cos^{2}x}\,dx & =\int\frac{1}{1+u^{2}}\,du\\
 & =\arctan(u)+C\\
 & =\arctan(-\cos x)+C
\end{align*}
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{\cos x}{4+\sin^{2}x}\,dx
\]
}%
\end{problem*}
\begin{sol*}
נציב \L{%
\[
t=\sin(x),\qquad dt=\cos(x)\,dx
\]
}%
\L{%
\[
\int\frac{\cos x}{4+\sin^{2}x}\,dx=\int\frac{1}{2^{2}+t^{2}}\,dt=\frac{1}{4}\int\frac{1}{1+\left(\frac{t}{2}\right)^{2}}\,dt
\]
}%

נשתמש שוב באינטגרציה בהצבה: \L{$s=\frac{t}{2}$}. אז \L{$dt=2ds$}
לכן \L{%
\begin{align*}
\frac{1}{4}\int\frac{1}{1+\left(\frac{t}{2}\right)^{2}}\,dt & =\frac{1}{4}\int\frac{2}{1+s^{2}}\,ds=\frac{1}{2}\arctan(s)+C\\
 & =\frac{1}{2}\arctan\!\left(\frac{\sin x}{2}\right)+C
\end{align*}
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{e^{x}}{1+e^{2x}}\,dx
\]
}%
\end{problem*}
\begin{sol*}
נציב \L{%
\[
t=e^{x},\qquad dt=e^{x}\,dx
\]
}%
\L{%
\begin{align*}
\int\frac{e^{x}}{1+e^{2x}}\,dx & =\int\frac{1}{1+t^{2}}\,dt=\arctan(e^{x})+C
\end{align*}
}%
\end{sol*}
\begin{problem*}
חשבו את האינטגרל הלא מסוים \L{%
\[
\int\frac{1}{e^{x}+e^{-x}}\,dx
\]
}%
\end{problem*}
\begin{sol*}
נכפיל את המונה והמכנה ב-\L{$e^{x}$}: \L{%
\begin{align*}
\int\frac{1}{e^{x}+e^{-x}}\,dx & =\int\frac{e^{x}}{e^{x}\left(e^{x}+e^{-x}\right)}\,dx=\int\frac{e^{x}}{e^{2x}+1}\,dx\\
 & =\arctan(e^{x})+C
\end{align*}
}%
\end{sol*}
\begin{problem*}
תהי \L{$f$} גזירה ב-\L{$\mathbb{R}$} כך ש- \L{%
\[
\forall x\in\mathbb{R}:\quad f'(x)=f(x)
\]
}%
הוכיחו שקיים \L{$c\in\mathbb{R}$} כך ש- \L{%
\[
f(x)=ce^{x}
\]
}%
\end{problem*}
\begin{proof}
נסתכל על \L{%
\[
g(x)=\frac{f(x)}{e^{x}}
\]
}%
אז \L{%
\[
g'(x)=\frac{f'(x)e^{x}-f(x)\left(e^{x}\right)^{'}}{e^{2x}}=\frac{f(x)e^{x}-f(x)e^{x}}{e^{2x}}=0
\]
}%
\L{$\Leftarrow$} לפי משפט לגראנז' \L{$g$} קבועה, כלומר קיים \L{$c\in\mathbb{R}$}
כך ש-\L{$g(x)=c$}

\L{$\Leftarrow$} \L{$f(x)=ce^{x}$}
\end{proof}
\begin{problem*}
מצאו את כל הפונקציות הקדומות של \L{%
\[
f(x)=\ln(x)\,\frac{\ln(\ln x)}{x}
\]
}%
ב-\L{$(1,\infty)$}.

נציב \L{%
\[
t=\ln(x),\qquad dt=\frac{dx}{x}
\]
}%
\L{%
\begin{align*}
\int\ln(x)\cdot\frac{\ln(\ln x)}{x}\,dx & =\int t\ln(t)\,dt=\int\ln(t)\cdot\left(\frac{t^{2}}{2}\right)'dt\\
 & =\frac{t^{2}\ln(t)}{2}-\int\frac{t^{2}}{2t}\,dt\\
 & =\frac{t^{2}\ln(t)}{2}-\frac{1}{2}\int t\,dt\\
 & =\frac{t^{2}\ln(t)}{2}-\frac{t^{2}}{4}+C\\
 & =\frac{\ln(x)^{2}\ln(\ln(x))}{2}-\frac{\ln(x)^{2}}{4}+C
\end{align*}
}%

מצאו את כל הפונקציות הקדומות של \L{%
\[
f(x)=\arctan x
\]
}%
ב-\L{$\mathbb{R}$}.
\end{problem*}
\begin{sol*}
מתקיים \L{%
\begin{align*}
\int1\cdot\arctan(x)\,dx & =\int x'\cdot\arctan(x)\,dx=x\arctan(x)-\int\frac{x}{1+x^{2}}\,dx
\end{align*}
}%
\L{%
\begin{align*}
\int\frac{x}{1+x^{2}}\,dx & =\frac{1}{2}\int\frac{\left(1+x^{2}\right)'}{1+x^{2}}\,dx=\frac{1}{2}\ln\left(1+x^{2}\right)+C
\end{align*}
}%
הביטוי \L{$1+x^{2}$} תמיד חיובי לכן אין צורך בערך מוחלט.
\end{sol*}

\part{הפונקציות \L{$\exp,\ln$}}
\begin{problem*}
הוכיחו שהגבול \L{%
\[
\lim_{n\to\infty}\left(\frac{n+2}{n-1}\right)^{n}
\]
}%
קיים וחשבו את ערכו.
\end{problem*}
\begin{sol*}
מתקיים\L{%
\begin{align*}
\lim_{n\to\infty}\left(\frac{n+2}{n-1}\right)^{n} & =\lim_{n\to\infty}\left(\frac{1+\frac{2}{n}}{1-\frac{1}{n}}\right)^{n}=\lim_{n\to\infty}\frac{\left(1+\frac{2}{n}\right)^{n}}{\left(1-\frac{1}{n}\right)^{n}}
\end{align*}
}%
לפי ההגדרה של \L{$\exp$}: \L{%
\[
\lim_{n\to\infty}\left(1+\frac{2}{n}\right)^{n}=\exp(2)=e^{2}
\]
}%
\L{%
\[
\lim_{n\to\infty}\left(1+\frac{-1}{n}\right)^{n}=\exp(-1)=e^{-1}
\]
}%
\L{$\Leftarrow$} מאריתמטיקה של גבולות \L{%
\[
\lim_{n\to\infty}\left(\frac{n+2}{n-1}\right)^{n}=\frac{e^{2}}{e^{-1}}\underset{e^{-1}\ne0}{=}e^{3}
\]
}%
\end{sol*}
\begin{problem*}
הוכיחו שלכל \L{$x\in\mathbb{R}$} מתקיים \L{%
\[
\exp(x)\ge1+x
\]
}%
והסיקו ש- \L{%
\[
\lim_{x\to\infty}\exp(x)=\infty
\]
}%
\end{problem*}
\begin{proof}
יהי \L{$x\in\mathbb{R}$}. לפי ההגדרה \L{%
\[
\exp(x)=\lim_{n\to\infty}\left(1+\frac{x}{n}\right)^{n}
\]
}%
הוכחנו בהרצאה בעזרת א\textquotedblright ש הממוצעים שהסדרה \L{$\left(\left(1+\frac{x}{n}\right)^{n}\right)_{n=1}^{\infty}$}
מונוטונית עולה החל ממקום מסוים. 

יהי \L{$N=\left|x\right|$}. אז לכל \L{$n>N$} מתקיים, לפי א\textquotedblright ש
ברנולי, \L{%
\[
1+x\le\left(1+\frac{x}{n}\right)^{n}
\]
}%
וגבולות משמרים באופן חלש א\textquotedblright ש של איברי הסדרה לכן
\L{%
\[
\exp(x)\ge1+x
\]
}%
בנוסף \L{%
\[
\lim_{x\to\infty}(1+x)=\infty
\]
}%
לכן לפי משפט הפרוסה ומה שהוכחנו \L{%
\[
\lim_{x\to\infty}\exp(x)=\infty
\]
}%
\end{proof}
\begin{problem*}
הוכיחו שלכל \L{$y_{1},y_{2}>0$} \L{%
\[
\ln(y_{1}y_{2})=\ln(y_{1})+\ln(y_{2})
\]
}%
והסיקו שלכל \L{$y>0$} \L{%
\[
\ln\!\left(\frac{1}{y}\right)=-\ln(y)
\]
}%
\end{problem*}
\begin{proof}
יהיו \L{$y_{1},y_{2}\in(0,\infty)$}. זו התמונה של \L{$\exp$}, לכן
קיימים \L{$x_{1},x_{2}\in\mathbb{R}$} כך ש- \L{%
\[
\exp(x_{1})=y_{1},\qquad\exp(x_{2})=y_{2}
\]
}%
הוכחנו שמתקיים \L{%
\[
\exp(x_{1}+x_{2})=\exp(x_{1})\cdot\exp(x_{2})=y_{1}y_{2}
\]
}%
\L{$\Leftarrow$} מהגדרת \L{$\ln$} כפונקציה ההפוכה של \L{$\exp$}
\L{%
\[
\ln(y_{1}y_{2})=x_{1}+x_{2}=\ln(y_{1})+\ln(y_{2})
\]
}%
בנוסף, עבור \L{$y>0$} קיים \L{$x\in\mathbb{R}$} כך ש-\L{$\exp(x)=y$}.
לכן \L{%
\begin{align*}
\exp(-x)=\exp(0-x)=\frac{\exp(0)}{\exp(x)}=\frac{1}{\exp(x)}=\frac{1}{y}
\end{align*}
}%
\L{%
\[
\Rightarrow\quad\ln\!\left(\frac{1}{y}\right)=-x=-\ln(y)
\]
}%
\end{proof}
